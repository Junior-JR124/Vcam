# This file installs the NVIDIA (CUDA 12) build of onnxruntime, which also
# contains the CPU provider, so the Device dropdown in vcam_ui.py (Auto / CPU /
# GPU) works with this single package. In "Auto" mode vcam time-trials the GPU
# at startup and only uses it if it clearly beats the CPU; on small GPUs the CPU
# is often just as fast, so Auto may well stay on CPU.
#
# Install ONLY ONE onnxruntime variant. Do not install onnxruntime or
# onnxruntime-directml alongside onnxruntime-gpu: they share one module name and
# conflict. If you pip-installed one before, uninstall it first:
#   pip uninstall -y onnxruntime onnxruntime-directml onnxruntime-gpu
onnxruntime-gpu==1.20.2

# CUDA 12 / cuDNN 9 runtime DLLs, installed from pip so no system CUDA Toolkit or
# cuDNN install is needed. Pinned to 12.6 to match NVIDIA driver 560 (CUDA 12.6);
# a newer driver (580+) can use newer 12.x builds. Do NOT use CUDA 13: it dropped
# support for Pascal cards such as the GTX 10-series.
# vcam.py must register the venv's site-packages\nvidia\*\bin folders with
# os.add_dll_directory() before onnxruntime is imported (see the top of vcam.py).
